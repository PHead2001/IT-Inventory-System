function useInventoryOwnedRef(ref, record, value) {
  useLayoutEffect(() => {
    if (!ref || !record || !record.current()) return;
    const token = {};
    inventoryPublishedRefs.set(ref, token);
    let cleanup;
    if (typeof ref === "function") cleanup = ref(value);
    else ref.current = value;
    return () => {
      // A retained cleanup cannot clear a newer publication of this same ref.
      if (inventoryPublishedRefs.get(ref) !== token) return;
      inventoryPublishedRefs.delete(ref);
      if (typeof cleanup === "function") cleanup();
      else if (typeof ref === "function") ref(null);
      else if (ref.current === value) ref.current = null;
    };
  }, [ref, record, value]);
}

function inventoryBuildPasses(nodes, camera, mergeMode, current) {
  const passes = [];
  try {
    for (let i = 0; i < nodes.length; i++) {
      if (!current()) break;
      const node = nodes[i];
      if (node instanceof Effect) {
        const effects = [node];
        let convolution = isConvolution(node), mainUv = hasMainUv(node);
        if (mergeMode !== "none") {
          let next;
          while ((next = nodes[i + 1]) instanceof Effect) {
            const nextConvolution = isConvolution(next), nextMainUv = hasMainUv(next);
            if (mergeMode === "auto" && (convolution && nextConvolution || convolution && nextMainUv || mainUv && nextConvolution)) break;
            effects.push(next);
            convolution ||= nextConvolution;
            mainUv ||= nextMainUv;
            i++;
          }
        }
        if (!current()) break;
        const pass = new EffectPass(camera, ...effects);
        generatedPasses.add(pass);
        passes.push(pass);
      } else if (node instanceof Pass) passes.push(node);
    }
    return passes;
  } catch (error) {
    for (const pass of passes) {
      try { inventoryDisposeGenerated(pass); } catch { /* Continue owned rollback. */ }
    }
    throw error;
  }
}
